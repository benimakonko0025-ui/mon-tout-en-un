    1 · Alertes sur ton téléphone
    1. Dans l'app MT5 du téléphone : Paramètres → Messages, note ton MetaQuotes ID.
    2. Sur ordinateur : Outils → Options → Notifications, coche « Activer les notifications Push » et colle l'ID.

    // Appelle LogSignal("BUY", prix, sl, tp, "Croisement MM20/MM50")
// à l'endroit où ton bot décide d'entrer.
void LogSignal(string sens, double prix, double sl, double tp, string raison)
{
   int h = FileOpen("signaux.csv",
                    FILE_READ|FILE_WRITE|FILE_CSV|FILE_ANSI|FILE_COMMON, ',');
   if(h == INVALID_HANDLE) { Print("Erreur fichier : ", GetLastError()); return; }
   if(FileSize(h) == 0)
      FileWrite(h, "date", "symbole", "sens", "prix", "sl", "tp", "raison");
   FileSeek(h, 0, SEEK_END);
   FileWrite(h, TimeToString(TimeCurrent(), TIME_DATE|TIME_MINUTES), _Symbol, sens,
             DoubleToString(prix, _Digits), DoubleToString(sl, _Digits),
             DoubleToString(tp, _Digits), raison);
   FileClose(h);
   SendNotification(sens + " " + _Symbol + " @ " + DoubleToString(prix, _Digits)
                    + " | " + raison);
}
    3 · Script ExportJournal.mq5

    Dans MetaEditor : Nouveau → Script, colle ce code, compile, puis glisse le script sur un graphique. Il crée journal.csv avec tes trades clôturés.

    #property script_show_inputs
input int Jours = 30;   // période à exporter

void OnStart()
{
   if(!HistorySelect(TimeCurrent() - Jours * 86400, TimeCurrent()))
   { Print("Historique indisponible"); return; }
   int total = HistoryDealsTotal();

   // 1) prix d'entrée de chaque position
   long ids[]; double prixIn[];
   for(int i = 0; i < total; i++)
   {
      ulong t = HistoryDealGetTicket(i);
      if(HistoryDealGetInteger(t, DEAL_ENTRY) != DEAL_ENTRY_IN) continue;
      int n = ArraySize(ids);
      ArrayResize(ids, n + 1); ArrayResize(prixIn, n + 1);
      ids[n]    = HistoryDealGetInteger(t, DEAL_POSITION_ID);
      prixIn[n] = HistoryDealGetDouble(t, DEAL_PRICE);
   }

   int h = FileOpen("journal.csv", FILE_WRITE|FILE_CSV|FILE_ANSI|FILE_COMMON, ',');
   if(h == INVALID_HANDLE) { Print("Erreur fichier : ", GetLastError()); return; }
   FileWrite(h, "date", "symbole", "sens", "lots", "entree", "sortie", "pnl");

   // 2) une ligne par clôture
   for(int i = 0; i < total; i++)
   {
      ulong t = HistoryDealGetTicket(i);
      if(HistoryDealGetInteger(t, DEAL_ENTRY) != DEAL_ENTRY_OUT) continue;
      long   pos = HistoryDealGetInteger(t, DEAL_POSITION_ID);
      double entree = 0;
      for(int k = 0; k < ArraySize(ids); k++) if(ids[k] == pos) { entree = prixIn[k]; break; }
      string sym = HistoryDealGetString(t, DEAL_SYMBOL);
      int    dg  = (int)SymbolInfoInteger(sym, SYMBOL_DIGITS);
      // une clôture en vente ferme un achat, et inversement
      string sens = (HistoryDealGetInteger(t, DEAL_TYPE) == DEAL_TYPE_SELL) ? "BUY" : "SELL";
      double pnl = HistoryDealGetDouble(t, DEAL_PROFIT) + HistoryDealGetDouble(t, DEAL_SWAP)
                 + HistoryDealGetDouble(t, DEAL_COMMISSION);
      FileWrite(h, TimeToString((datetime)HistoryDealGetInteger(t, DEAL_TIME), TIME_DATE|TIME_MINUTES),
                sym, sens, DoubleToString(HistoryDealGetDouble(t, DEAL_VOLUME), 2),
                DoubleToString(entree, dg), DoubleToString(HistoryDealGetDouble(t, DEAL_PRICE), dg),
                DoubleToString(pnl, 2));
   }
   FileClose(h);
   Print("journal.csv créé dans le dossier Common\\Files");
}
    Où trouver les fichiers
    1. signaux.csv et journal.csv : dans MT5, Fichier → Ouvrir le dossier des données, remonte d'un dossier, puis Common → Files.
    2. Bougies pour les indicateurs : Affichage → Symboles (Ctrl+U), onglet Barres, choisis la paire et la période, Demander, puis Exporter les barres.
    3. Importe chaque fichier dans l'onglet correspondant. Tu peux ouvrir ce tableau de bord sur ordinateur pour importer, puis le consulter sur ton téléphone : les données sont les mêmes partout.
